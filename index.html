// Fungsi Ekspor Data Rekapitulasi ke Format CSV
function exportRekapToCSV(dataArray, filename = 'laporan_rekap.csv') {
  const headers = ['ID', 'Nama Mahasiswa', 'NIM', 'Nilai', 'Status'];
  const rows = dataArray.map(item => [
    item.id,
    `"${item.nama}"`,
    item.nim,
    item.nilai,
    item.status
  ]);
  
  const csvContent = [headers.join(','), ...rows.map(e => e.join(','))].join('\n');
  const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
  const link = document.createElement('a');
  link.href = URL.createObjectURL(blob);
  link.setAttribute('download', filename);
  link.click();
}

Implementasi OpenCV & MediaPipe (Python)
import cv2
import mediapipe as mp

mp_hands = mp.solutions.hands
hands = mp_hands.Hands(max_num_hands=2, min_detection_confidence=0.7)
mp_draw = mp.solutions.drawing_utils
cap = cv2.VideoCapture(0)

while cap.isOpened():
    success, img = cap.read()
    if not success: break
    
    img_rgb = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
    results = hands.process(img_rgb)
    
    if results.multi_hand_landmarks:
        for hand_landmarks in results.multi_hand_landmarks:
            mp_draw.draw_landmarks(img, hand_landmarks, mp_hands.HAND_CONNECTIONS)
            
    cv2.imshow('Hand Tracking', img)
    if cv2.waitKey(1) & 0xFF == ord('q'): break

cap.release()
cv2.destroyAllWindows()

Implementasi Dark Mode Logic (JavaScript)
// Fungsi Toggle Dark Mode & LocalStorage Memory
function applyTheme(isDark) {
  const htmlElement = document.documentElement;
  if (isDark) {
    htmlElement.setAttribute('data-bs-theme', 'dark');
    document.body.classList.add('dark-theme');
    localStorage.setItem('reza_theme', 'dark');
  } else {
    htmlElement.setAttribute('data-bs-theme', 'light');
    document.body.classList.remove('dark-theme');
    localStorage.setItem('reza_theme', 'light');
  }
}
